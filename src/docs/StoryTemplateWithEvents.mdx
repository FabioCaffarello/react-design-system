# Story Template com Eventos e Estados

Este documento fornece um template e guia para criar stories que documentam eventos e estados dos componentes.

## Template Básico

```tsx
import type { Meta, StoryObj } from '@storybook/react';
import { useState } from 'react';
import { expect, userEvent, within, waitFor } from '@storybook/test';
import { fn } from '@storybook/test';
import YourComponent from './YourComponent';

const meta: Meta<typeof YourComponent> = {
  title: 'Category/YourComponent',
  component: YourComponent,
  tags: ['autodocs'],
  parameters: {
    docs: {
      description: {
        component: 'Description of your component and its purpose.',
      },
    },
  },
  argTypes: {
    // Document all props
    propName: {
      description: 'Description of the prop',
      control: 'text', // or 'boolean', 'select', etc.
      table: {
        type: { summary: 'string' },
        defaultValue: { summary: "'default value'" },
      },
    },
    onEvent: {
      description: 'Callback fired when event occurs',
      action: 'onEvent',
      table: {
        type: { summary: '(param: Type) => void' },
      },
    },
  },
};

export default meta;
type Story = StoryObj<typeof YourComponent>;

// Default story with interaction testing
export const Default: Story = {
  render: () => {
    const [state, setState] = useState(initialState);
    const handleEvent = fn((param: Type) => {
      setState(newState);
    });
    
    return (
      <YourComponent
        prop={value}
        onEvent={handleEvent}
      />
    );
  },
  play: async ({ canvasElement }) => {
    const canvas = within(canvasElement);
    
    // Interact with component
    const button = canvas.getByRole('button', { name: /label/i });
    await userEvent.click(button);
    
    // Verify state change
    await waitFor(() => {
      expect(canvas.getByText(/expected result/i)).toBeInTheDocument();
    });
  },
};
```

## Documentando Eventos

### 1. Usando `fn()` do @storybook/test

```tsx
import { fn } from '@storybook/test';

export const WithEvent: Story = {
  render: () => {
    const handleClick = fn((id: string) => {
      console.log('Clicked:', id);
    });
    
    return (
      <YourComponent
        onClick={handleClick}
      />
    );
  },
};
```

### 2. Documentando no argTypes

```tsx
argTypes: {
  onOpenChange: {
    description: 'Callback fired when the open state changes',
    action: 'onOpenChange', // This will show in Actions panel
    table: {
      type: { summary: '(open: boolean) => void' },
    },
  },
}
```

### 3. Testando Eventos com play functions

```tsx
play: async ({ canvasElement }) => {
  const canvas = within(canvasElement);
  const button = canvas.getByRole('button');
  
  await userEvent.click(button);
  
  // Verify event was called (if using fn())
  // The action will appear in Actions panel automatically
};
```

## Documentando Estados

### 1. Stories para Cada Estado

```tsx
export const DefaultState: Story = {
  args: {
    status: 'default',
  },
};

export const ActiveState: Story = {
  args: {
    status: 'active',
  },
};

export const ErrorState: Story = {
  args: {
    status: 'error',
  },
};
```

### 2. Stories com Transições de Estado

```tsx
export const StateTransitions: Story = {
  render: () => {
    const [status, setStatus] = useState<'default' | 'active' | 'error'>('default');
    
    return (
      <div>
        <YourComponent status={status} />
        <button onClick={() => setStatus('active')}>Activate</button>
        <button onClick={() => setStatus('error')}>Error</button>
      </div>
    );
  },
  play: async ({ canvasElement }) => {
    const canvas = within(canvasElement);
    
    // Test state transition
    await userEvent.click(canvas.getByRole('button', { name: /activate/i }));
    
    await waitFor(() => {
      expect(canvas.getByText(/active/i)).toBeInTheDocument();
    });
  },
};
```

## Exemplos Completos

### Exemplo 1: CommandPalette com Eventos

```tsx
export const KeyboardNavigation: Story = {
  render: () => {
    const [open, setOpen] = useState(false);
    const handleOpenChange = fn((newOpen: boolean) => {
      setOpen(newOpen);
    });
    
    return (
      <div>
        <Button onClick={() => setOpen(true)}>Open</Button>
        <CommandPalette
          items={items}
          open={open}
          onOpenChange={handleOpenChange}
        />
      </div>
    );
  },
  play: async ({ canvasElement }) => {
    const canvas = within(canvasElement);
    
    // Open command palette
    await userEvent.click(canvas.getByRole('button', { name: /open/i }));
    
    // Wait for it to open
    await waitFor(() => {
      expect(canvas.getByPlaceholderText(/type a command/i)).toBeInTheDocument();
    });
    
    // Test keyboard navigation
    const input = canvas.getByPlaceholderText(/type a command/i);
    await userEvent.type(input, 'save');
    await userEvent.keyboard('{ArrowDown}');
    await userEvent.keyboard('{Enter}');
    
    // Verify it closed
    await waitFor(() => {
      expect(canvas.queryByPlaceholderText(/type a command/i)).not.toBeInTheDocument();
    });
  },
};
```

### Exemplo 2: Toast com Estados

```tsx
export const AllVariants: Story = {
  render: () => {
    const toast = useToast();
    
    return (
      <div>
        <Button onClick={() => toast.success('Success', 'Message')}>
          Success
        </Button>
        <Button onClick={() => toast.error('Error', 'Message')}>
          Error
        </Button>
        <ToastContainer />
      </div>
    );
  },
  play: async ({ canvasElement }) => {
    const canvas = within(canvasElement);
    
    // Show all variants
    await userEvent.click(canvas.getByRole('button', { name: /success/i }));
    await userEvent.click(canvas.getByRole('button', { name: /error/i }));
    
    // Wait for toasts to appear (entering state)
    await waitFor(() => {
      expect(canvas.getByText(/success/i)).toBeInTheDocument();
      expect(canvas.getByText(/error/i)).toBeInTheDocument();
    });
    
    // Wait for auto-dismiss (exiting state)
    await waitFor(
      () => {
        expect(canvas.queryByText(/success/i)).not.toBeInTheDocument();
      },
      { timeout: 5000 }
    );
  },
};
```

## Boas Práticas

### 1. Sempre Use `fn()` para Event Handlers

```tsx
// ✅ Good
const handleClick = fn(() => {
  // handler logic
});

// ❌ Bad
const handleClick = () => {
  // handler logic
};
```

### 2. Teste Interações Reais

```tsx
// ✅ Good - tests real interaction
play: async ({ canvasElement }) => {
  const canvas = within(canvasElement);
  await userEvent.click(canvas.getByRole('button'));
};

// ❌ Bad - just renders
export const Default: Story = {
  args: {
    onClick: () => {},
  },
};
```

### 3. Use `waitFor` para Estados Assíncronos

```tsx
// ✅ Good
await waitFor(() => {
  expect(canvas.getByText(/result/i)).toBeInTheDocument();
});

// ❌ Bad
expect(canvas.getByText(/result/i)).toBeInTheDocument();
```

### 4. Documente Todos os Estados

```tsx
// ✅ Good - separate stories for each state
export const DefaultState: Story = { /* ... */ };
export const ActiveState: Story = { /* ... */ };
export const ErrorState: Story = { /* ... */ };

// ❌ Bad - only one story
export const Default: Story = { /* ... */ };
```

### 5. Documente Eventos no argTypes

```tsx
// ✅ Good
argTypes: {
  onOpenChange: {
    description: 'Callback fired when open state changes',
    action: 'onOpenChange',
    table: {
      type: { summary: '(open: boolean) => void' },
    },
  },
}
```

## Checklist para Stories

- [ ] Story tem descrição clara
- [ ] Todos os props têm argTypes
- [ ] Eventos são documentados
- [ ] Estados são documentados
- [ ] Story tem teste de interação (play function)
- [ ] Teste cobre fluxo principal
- [ ] Usa `fn()` para event handlers
- [ ] Usa `waitFor` para estados assíncronos
- [ ] Story demonstra comportamento real
